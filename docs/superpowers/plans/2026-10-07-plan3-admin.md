### Task 1: Admin layout with role guard

**Files:**
- Create: `app/admin/layout.tsx`

- [ ] **Step 1: Create `app/admin/layout.tsx`**

This is a Server Component. It reads the access token, calls `/users/me`, and redirects if the user is not ADMIN or STAFF.

```tsx
import { redirect } from "next/navigation";
import { cookies } from "next/headers";
import Link from "next/link";
import { usersApi } from "@/lib/api/users";
import { ACCESS_TOKEN_COOKIE } from "@/lib/auth/cookies";

const NAV = [
  { href: "/admin", label: "Dashboard" },
  { href: "/admin/products", label: "Products" },
  { href: "/admin/categories", label: "Categories" },
  { href: "/admin/orders", label: "Orders" },
  { href: "/admin/users", label: "Users" },
  { href: "/admin/coupons", label: "Coupons" },
];

export default async function AdminLayout({ children }: LayoutProps<"/">) {
  const cookieStore = await cookies();
  const hasToken = Boolean(cookieStore.get(ACCESS_TOKEN_COOKIE)?.value);

  if (!hasToken) redirect("/login?next=/admin");

  let user;
  try {
    user = await usersApi.getMe();
  } catch {
    redirect("/login?next=/admin");
  }

  if (user.role !== "ADMIN" && user.role !== "STAFF") {
    redirect("/");
  }

  return (
    <div className="mx-auto max-w-7xl px-5 py-10 sm:px-8">
      <div className="mb-8 flex items-center justify-between">
        <h1 className="text-[11px] font-medium uppercase tracking-[0.28em]">Admin</h1>
        <span className="text-[11px] uppercase tracking-wider text-mute">{user.role}</span>
      </div>
      <div className="grid grid-cols-1 gap-10 lg:grid-cols-[180px_1fr]">
        <nav className="flex flex-row flex-wrap gap-4 lg:flex-col">
          {NAV.map((item) => (
            <Link
              key={item.href}
              href={item.href}
              className="text-[12px] uppercase tracking-[0.2em] text-mute transition-colors hover:text-foreground"
            >
              {item.label}
            </Link>
          ))}
        </nav>
        <main>{children}</main>
      </div>
    </div>
  );
}
```

"use client";
import { useCallback, useEffect, useState } from "react";
import { adminProductsApi } from "@/lib/api/admin/products";
import { adminCategoriesApi } from "@/lib/api/admin/categories";
import { adminInventoryApi } from "@/lib/api/admin/inventory";
import { DataTable, type Column } from "@/components/ui/data-table";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Badge } from "@/components/ui/badge";
import { ApiError } from "@/lib/api/errors";
import type { Product, Category, Gender, ProductCondition } from "@/lib/api/types";

type BadgeVariant = "default" | "success" | "warning" | "error" | "info";

function statusVariant(status: string): BadgeVariant {
  const map: Record<string, BadgeVariant> = {
    PUBLISHED: "success",
    DRAFT: "warning",
    ARCHIVED: "default",
    SOLD: "info",
  };
  return map[status] ?? "default";
}

function formatInr(amount: number) {
  return new Intl.NumberFormat("en-IN", { style: "currency", currency: "INR", maximumFractionDigits: 0 }).format(amount);
}

const GENDERS: Gender[] = ["WOMEN", "MEN", "UNISEX", "KIDS"];
const CONDITIONS: ProductCondition[] = ["NEW_WITH_TAGS", "EXCELLENT", "GOOD", "FAIR"];

function ProductForm({
  initial,
  categories,
  onSave,
  onCancel,
}: {
  initial?: Product;
  categories: Category[];
  onSave: () => void;
  onCancel: () => void;
}) {
  const [form, setForm] = useState({
    name: initial?.name ?? "",
    description: initial?.description ?? "",
    brand: initial?.brand ?? "",
    category: typeof initial?.category === "object" ? initial.category.id : (initial?.category ?? ""),
    gender: (initial?.gender ?? "WOMEN") as Gender,
    size: initial?.size ?? "",
    color: initial?.color ?? "",
    material: initial?.material ?? "",
    condition: (initial?.condition ?? "EXCELLENT") as ProductCondition,
    conditionDescription: initial?.conditionDescription ?? "",
    originalPrice: initial?.originalPrice ?? 0,
    sellingPrice: initial?.sellingPrice ?? 0,
    sku: initial?.sku ?? "",
    featured: initial?.featured ?? false,
  });
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState("");

  function setField<K extends keyof typeof form>(key: K, value: (typeof form)[K]) {
    setForm((p) => ({ ...p, [key]: value }));
  }

  async function handleSubmit(e: React.FormEvent) {
    e.preventDefault();
    setError(""); setLoading(true);
    try {
      if (initial) {
        await adminProductsApi.update(initial.id, {
          ...form,
          originalPrice: Number(form.originalPrice),
          sellingPrice: Number(form.sellingPrice),
        });
      } else {
        const product = await adminProductsApi.create({
          ...form,
          originalPrice: Number(form.originalPrice),
          sellingPrice: Number(form.sellingPrice),
        });
        await adminInventoryApi.init(product.id, { quantity: 1, reason: "Initial stock" });
      }
      onSave();
    } catch (err) {
      setError(err instanceof ApiError ? err.message : "Save failed.");
    } finally {
      setLoading(false);
    }
  }

  return (
    <form onSubmit={handleSubmit} className="flex flex-col gap-4 border border-line p-5 mt-4">
      <div className="grid grid-cols-2 gap-4">
        <Input label="Name" id="name" required value={form.name} onChange={(e) => setField("name", e.target.value)} />
        <Input label="Brand" id="brand" required value={form.brand} onChange={(e) => setField("brand", e.target.value)} />
      </div>
      <div>
        <label className="mb-1.5 block text-[11px] font-medium uppercase tracking-[0.2em] text-mute">Description</label>
        <textarea
          required
          value={form.description}
          onChange={(e) => setField("description", e.target.value)}
          rows={3}
          className="w-full border border-line bg-transparent px-3 py-2.5 text-sm outline-none focus:border-foreground"
        />
      </div>
      <div className="grid grid-cols-2 gap-4">
        <div>
          <label className="mb-1.5 block text-[11px] font-medium uppercase tracking-[0.2em] text-mute">Category</label>
          <select value={form.category} onChange={(e) => setField("category", e.target.value)} required className="w-full border border-line bg-transparent px-3 py-2.5 text-sm">
            <option value="">Select…</option>
            {categories.map((c) => <option key={c.id} value={c.id}>{c.name}</option>)}
          </select>
        </div>
        <div>
          <label className="mb-1.5 block text-[11px] font-medium uppercase tracking-[0.2em] text-mute">Gender</label>
          <select value={form.gender} onChange={(e) => setField("gender", e.target.value as Gender)} className="w-full border border-line bg-transparent px-3 py-2.5 text-sm">
            {GENDERS.map((g) => <option key={g} value={g}>{g}</option>)}

        <Input label="Material" id="material" value={form.material} onChange={(e) => setField("material", e.target.value)} />
      </div>
      <div className="grid grid-cols-2 gap-4">
        <div>

      <div className="flex gap-3">
        <Button type="submit" loading={loading}>{initial ? "Update" : "Create"}</Button>
        <Button type="button" variant="ghost" onClick={onCancel}>Cancel</Button>
      </div>
    </form>
  );
}

export default function AdminProductsPage() {
  const [products, setProducts] = useState<Product[]>([]);
  const [categories, setCategories] = useState<Category[]>([]);
  const [loading, setLoading] = useState(true);
  const [page, setPage] = useState(1);
  const [totalPages, setTotalPages] = useState(1);
  const [creating, setCreating] = useState(false);
  const [editing, setEditing] = useState<Product | null>(null);
  const [actionError, setActionError] = useState("");

  const load = useCallback(async (p: number) => {
    setLoading(true);
    try {
      const result = await adminProductsApi.list({ page: p, limit: 20 });
      setProducts(result.data);
      setTotalPages(result.pagination.totalPages);
    } finally {
      setLoading(false);
    }
  }, []);

  useEffect(() => { void load(page); }, [load, page]);

  async function loadCategories() {
    const { categoriesApi } = await import("@/lib/api/categories");
    setCategories(await categoriesApi.list());
  }

  useEffect(() => { void loadCategories(); }, []);

  async function handlePublish(id: string) {
    setActionError("");
    try { await adminProductsApi.publish(id); await load(page); }
    catch (err) { setActionError(err instanceof ApiError ? err.message : "Action failed."); }
  }

  async function handleArchive(id: string) {
    setActionError("");
    try { await adminProductsApi.archive(id); await load(page); }
    catch (err) { setActionError(err instanceof ApiError ? err.message : "Action failed."); }
  }

  async function handleDelete(id: string) {
    if (!confirm("Delete this product?")) return;
    setActionError("");
    try { await adminProductsApi.delete(id); await load(page); }
    catch (err) { setActionError(err instanceof ApiError ? err.message : "Delete failed."); }
  }

  const columns: Column<Product>[] = [
    { key: "name", header: "Name" },
    { key: "brand", header: "Brand" },
    { key: "sellingPrice", header: "Price", render: (row) => formatInr(row.sellingPrice as number) },
    { key: "status", header: "Status", render: (row) => <Badge variant={statusVariant(row.status as string)}>{(row.status as string).replace(/_/g, " ")}</Badge> },
    {
      key: "actions",
      header: "Actions",
      render: (row) => (
        <div className="flex gap-3 text-[12px]">
          <button onClick={() => setEditing(row)} className="text-mute hover:text-foreground">Edit</button>
          {row.status === "DRAFT" && <button onClick={() => handlePublish(row.id as string)} className="text-mute hover:text-foreground">Publish</button>}
          {row.status === "PUBLISHED" && <button onClick={() => handleArchive(row.id as string)} className="text-mute hover:text-foreground">Archive</button>}
          <button onClick={() => handleDelete(row.id as string)} className="text-red-500 hover:text-red-700">Delete</button>
        </div>
      ),
    },
  ];

  return (
    <div className="flex flex-col gap-6">
      <div className="flex items-center justify-between">
        <h2 className="text-[11px] uppercase tracking-[0.2em] text-mute">Products</h2>
        {!creating && !editing && (
          <Button variant="secondary" onClick={() => setCreating(true)}>+ New Product</Button>
        )}
      </div>

      {actionError && <p className="text-sm text-red-600">{actionError}</p>}

      {creating && (
        <ProductForm
          categories={categories}
          onSave={() => { setCreating(false); void load(page); }}
          onCancel={() => setCreating(false)}
        />
      )}

      {editing && (
        <ProductForm
          initial={editing}
          categories={categories}
          onSave={() => { setEditing(null); void load(page); }}
          onCancel={() => setEditing(null)}
        />
      )}

      <DataTable
        columns={columns}
        data={products as unknown as Record<string, unknown>[]}
        loading={loading}
        emptyMessage="No products found."
        page={page}
        totalPages={totalPages}
        onPageChange={setPage}
      />
    </div>
  );
}

import type { Order, OrderStatus } from "@/lib/api/types";

type BadgeVariant = "default" | "success" | "warning" | "error" | "info";

function statusVariant(status: OrderStatus): BadgeVariant {
  const map: Record<OrderStatus, BadgeVariant> = {
    PENDING_PAYMENT: "warning",
    PAYMENT_FAILED: "error",
    CONFIRMED: "info",
    PROCESSING: "info",
    SHIPPED: "info",

    CANCELLED: "error",
    REFUNDED: "default",
  };
  return map[status] ?? "default";
}

function formatInr(amount: number) {
  return new Intl.NumberFormat("en-IN", { style: "currency", currency: "INR", maximumFractionDigits: 0 }).format(amount);
}

function StatusUpdater({

function CouponForm({
  initial,

  initial?: Coupon;
  onSave: () => void;
  onCancel: () => void;
}) {
  const [form, setForm] = useState({

    description: initial?.description ?? "",

        expiryDate: new Date(form.expiryDate).toISOString(),

        await adminCouponsApi.create({ ...data, code: form.code });
